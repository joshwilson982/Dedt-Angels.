-- Debt Angels 4.4 -- Public simulated donation totals CREATE OR REPLACE VIEW public.demo_donation_totals WITH (security_invoker = true) AS SELECT a.id AS application_id, COALESCE(SUM(d.amount), 0)::numeric AS total_demo_donations, COUNT(d.id)::bigint AS contribution_count FROM public.debt_applications AS a LEFT JOIN public.demo_donations AS d ON d.application_id = a.id WHERE a.status = 'approved' GROUP BY a.id; -- Keep donor records private. -- Do not grant public access to individual donations. REVOKE ALL ON public.demo_donation_totals FROM PUBLIC, anon; GRANT SELECT ON public.demo_donation_totals TO authenticated; -- Check the totals using SQL Editor. SELECT * FROM public.demo_donation_totals;
